# 🌱 种子分类器使用指南

## 📋 概述

本项目新增了三个独立的种子分类器模型，用于二分类任务（背景/种子识别）：

- **EfficientNetB0Classifier** - 推荐使用，99%+准确率，15.3MB
- **ResNet18Classifier** - 快速推理，99%+准确率，42.6MB  
- **CustomCNNClassifier** - 轻量级自定义CNN（需要训练权重）

## 🚀 快速使用

### 1. 基本使用示例

```python
from utils.model_loader import ModelLoader
from PIL import Image

# 创建模型加载器
loader = ModelLoader(device='cpu')  # 或 'cuda'

# 加载分类器模型
success = loader.load_classifier_model('EfficientNetB0Classifier')

if success:
    # 加载和预处理图像
    image = Image.open('your_image.jpg').convert('RGB')
    image_tensor = loader.preprocess_image(image, size=224)
    
    # 进行分类预测
    result = loader.classify(image_tensor)
    
    print(f"预测类别: {result['predicted_class']}")
    print(f"置信度: {result['confidence']:.4f}")
    print(f"概率分布: {result['probabilities']}")
    
    # 卸载模型
    loader.unload_model()
```

### 2. 批量处理示例

```python
import os
from pathlib import Path

def batch_classify_images(image_dir, model_name='EfficientNetB0Classifier'):
    """批量分类图像"""
    loader = ModelLoader(device='cpu')
    
    # 加载模型
    if not loader.load_classifier_model(model_name):
        print(f"模型加载失败: {model_name}")
        return
    
    results = []
    image_files = list(Path(image_dir).glob('*.jpg')) + list(Path(image_dir).glob('*.png'))
    
    for image_path in image_files:
        try:
            # 加载图像
            image = Image.open(image_path).convert('RGB')
            image_tensor = loader.preprocess_image(image)
            
            # 分类
            result = loader.classify(image_tensor)
            result['image_path'] = str(image_path)
            results.append(result)
            
            print(f"{image_path.name}: {result['predicted_class']} ({result['confidence']:.3f})")
            
        except Exception as e:
            print(f"处理 {image_path} 失败: {e}")
    
    loader.unload_model()
    return results

# 使用示例
results = batch_classify_images('tests/images/')
```

### 3. 模型对比测试

```python
def compare_classifiers(image_path):
    """对比不同分类器的性能"""
    models = ['EfficientNetB0Classifier', 'ResNet18Classifier']
    
    image = Image.open(image_path).convert('RGB')
    
    for model_name in models:
        loader = ModelLoader(device='cpu')
        
        if loader.load_classifier_model(model_name):
            image_tensor = loader.preprocess_image(image)
            result = loader.classify(image_tensor)
            
            print(f"{model_name}:")
            print(f"  预测: {result['predicted_class']}")
            print(f"  置信度: {result['confidence']:.4f}")
            
            loader.unload_model()

# 使用示例
compare_classifiers('tests/images/100.jpg')
```

## 🔧 高级用法

### 1. 直接使用模型类

```python
from models import EfficientNetB0Classifier
import torch

# 创建模型实例
model = EfficientNetB0Classifier(num_classes=2, device='cpu')

# 加载权重
model.load_model_weight('weights/EfficientNetB0Classifier.pth')
model.eval()

# 预测
with torch.no_grad():
    # image_tensor: [1, 3, 224, 224]
    result = model.predict(image_tensor)
    print(result)
```

### 2. 获取模型信息

```python
# 获取详细模型信息

print(f"模型大小: {info['model_size_mb']:.1f}MB")

```

### 3. 自定义预处理

```python
import torchvision.transforms as transforms

# 自定义预处理管道
custom_transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.485, 0.456, 0.406], 
                        std=[0.229, 0.224, 0.225])
])

# 应用预处理
image_tensor = custom_transform(image).unsqueeze(0)
```

## 🚨 注意事项

1. **输入格式**: 图像必须是RGB格式，推荐尺寸224x224
2. **设备选择**: 支持CPU和CUDA，自动选择最佳设备
3. **内存管理**: 使用完毕后调用`unload_model()`释放内存
4. **权重文件**: 确保权重文件在`weights/`目录下
5. **依赖版本**: 需要PyTorch 1.8+和torchvision
